  \posterboxfixedcolor{Cell types are underspecified}{\ProblemTopBoxHeight}{problemline}{problembg}{%
    {\SmallBodyFont Marker genes give cell type labels meaning, but labels and markers do not always describe the same cells. Literature-extracted profiles reveal the failure modes: different names with the same markers, the same name with different markers, and partial overlaps that may reflect subtype, state, or context.\par}

  \posterboxfixedcolor{The \texttt{LLMarkers} dataset}{\ProblemBottomBoxHeight}{problemline}{problembg}{%
    {\SmallBodyFont We built \texttt{LLMarkers}, a human-curated benchmark of 2{,}528 reported markers from seven scRNA-seq studies. It contains 168 cell types, 641 Ensembl-mapped genes, and 1{,}560 unique cell type--gene pairs linked to source evidence and differentially expressed gene (DEG) comparisons. Most pairs were figure-only (74\%), with 16\% text-only and 10\% in both.\par}

    \vspace{0.20in}
    \postercenter{\llmarkersBenchmarkTable}
    \vspace{0.07in}
    {\TinyBodyFont\justifying\textbf{Table 1.} \texttt{LLMarkers} contains reported marker profiles from seven single-cell RNA-seq papers, split by source modality and DEG-table overlap.\par}
  }

  \posterboxfixedcolorcompact{\texttt{LLMarkersDB} enables text-searchable markers}{\ProblemDbBoxHeight}{problemline}{problembg}{%
    \texttt{LLMarkersDB} embeds source sentences and paper context with \texttt{sentence-transformers/all-MiniLM-L6-v2} and retrieves profiles by cosine similarity. Profiles also store Ensembl gene IDs, enabling exact gene lookup and Jaccard gene-set search across papers. Search the database at \href{https://sina.bio/llmarkers}{sina.bio/llmarkers}.
  }

  \posterboxfixedcolor{Evaluating expert marker curation}{\BenchmarkTopBoxHeight}{benchmarkline}{benchmarkbg}{%
    {\SmallBodyFont Using \texttt{LLMarkers}, we asked how expert-reported markers relate to differential expression rank. We swept the top $N$ ranked differentially expressed genes (DEGs) as predicted markers. Reported markers were enriched among strong DEGs, but rank cutoffs recovered only about half of them (mean peak F1: 0.41 text, 0.50 image).\par}

  \posterboxfixedcolor{Evaluating LLM marker curation}{\BenchmarkBottomBoxHeight}{benchmarkline}{benchmarkbg}{%
    {\SmallBodyFont We tested whether \texttt{claude-opus-4-6} could produce machine-readable marker records with structured cell type, gene, source sentence, and data ID fields. Extraction performed best (mean F1 0.67 for cell type--gene recovery; 0.58 when the correct data source was also required), outperforming generation (0.34) and selection (0.16).\par}

  \posterboxfixedcolor{Resolving cell type underspecification}{\AtlasBottomBoxHeight}{atlasline}{atlasbg}{%
    {\SmallBodyFont We used the large-scale corpus to return to the ambiguity in Figure 1. Each profile was encoded as a binary Ensembl gene vector and compared across papers by label and marker-gene Jaccard similarity. We then scored genes by coverage and purity within label- and marker-defined groups to nominate stable cell-type markers and context-dependent state markers.\par}
